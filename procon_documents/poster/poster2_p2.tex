  \newcommand{\ucrow}[2]{%
    \begin{minipage}[t]{0.25\linewidth}{\bfseries\fontsize{24pt}{29pt}\selectfont #1}\end{minipage}\hfill
    \begin{minipage}[t]{0.74\linewidth}{\fontsize{21pt}{29pt}\selectfont #2}\end{minipage}\par

  \newcommand{\ucbox}[4]{% 地の色, 文字色, 見出し, 行
    \begin{minipage}[t]{0.488\linewidth}\vspace{0pt}
      \tint{#1}{%
        {\fontsize{32pt}{38pt}\selectfont\bfseries\color{#2}#3}\par
        \vspace{8pt}
        \def\ucto{\ \textcolor{#2}{→}\ }#4}
    \end{minipage}}
