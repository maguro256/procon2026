  % ---- 4 つの要素を 2×2 のカードで
  \newcommand{\card}[2]{%
    \coloredbox[bgcolor=paper, fgcolor=ink, framecolor=paper, roundedcorners=6, linewidth=0pt]{%
      \head{#1}#2}}
  \begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
    \card{文脈：タスクと人の特徴}{%
      \centerline{$\x = \bigl[\,y,\ d,\ y d,\ \bm{e}_{\mathrm{diff}},\ \bm{e}_{\mathrm{perm}},\ \bm{e}_{\mathrm{eq}}\,\bigr]$}
      \begin{itemize}
        \item $y$：勤続年数、$d$：難易度
        \item $\bm{e}$：難易度・資格・機材の one-hot
        \item 「溶接が得意」「旋盤が得意」を区別
      \end{itemize}}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
    \card{報酬：平均と比べた速さ}{%
      \centerline{$r = c_f \cdot \bar t / t, \quad c_f \in \{1.5,\ 1.0,\ 0.5\}$}
      \begin{itemize}
        \item $t$：1 個あたりの所要時間
        \item $\bar t$：同じ難易度の過去の平均
        \item $c_f$：体感の簡単・普通・難しい
      \end{itemize}}
  \end{minipage}\par\vspace{8pt}
  \begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
    \card{学習：ベイズ線形回帰\,[4,5]}{%
      \centerline{$r \approx \thetab_w^{\top}\x, \quad \thetab_w \sim \mathcal N(\bm\mu_w, \Sigma_w)$}
      \begin{itemize}
        \item 作業者ごとに得意・不得意を学ぶ
        \item 更新は足し算 → ログを古い順に再生して組み立て直す
      \end{itemize}}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
    \card{選択：Thompson Sampling\,[3,6]}{%
      \centerline{$s_w = \tilde\thetab_w^{\top}\x \;\big/\; \bigl(1 + L_w / \hat T_j\bigr)$}
      \begin{itemize}
        \item $L_w$：手持ち、$\hat T_j$：見積もり時間
        \item 手持ちで割って負荷分散
      \end{itemize}}
  \end{minipage}
  \vspace{10pt}

\block{\num{5}シミュレーションによる評価}{%
  \begin{minipage}[t]{0.25\linewidth}
    \head{設定}
    \begin{itemize}
      \item 作業者 6 名、勤続 1〜30 年。真の時間倍率は AI に隠す
